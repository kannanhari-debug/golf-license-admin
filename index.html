Maintenance

Use these SQL templates in Supabase SQL Editor. They delete older data only.
Delete events older than 30 days
DELETE FROM events
WHERE created_at < now() - interval '30 days';
Delete sessions older than 30 days
DELETE FROM sessions
WHERE start_time < now() - interval '30 days';
Delete oldest 100 events (keep latest)
DELETE FROM events
WHERE id IN (
  SELECT id FROM events
  ORDER BY created_at ASC
  LIMIT 100
);
Delete oldest 100 sessions (keep latest)
DELETE FROM sessions
WHERE id IN (
  SELECT id FROM sessions
  ORDER BY start_time ASC
  LIMIT 100
);
Clear daily aggregates older than 90 days
DELETE FROM daily_usage
WHERE day < current_date - interval '90 days';

DELETE FROM daily_usage_named
WHERE day < current_date - interval '90 days';

DELETE FROM daily_attempts
WHERE day < current_date - interval '90 days';

DELETE FROM daily_attempts_named
WHERE day < current_date - interval '90 days';
